# ── เพิ่มใน backend/.env (ทุกเครื่อง/ทุก worker ต้องใช้ค่าเดียวกัน) ─────────────────────────────

# ความลับสำหรับเซ็น token — "ต้องตั้ง" เป็นค่าสุ่มยาว ≥ 32 ตัวอักษร (เช่น node -e "console.log(require('crypto').randomBytes(48).toString('hex'))")
# ถ้าไม่ตั้ง ระบบจะอนุมานจากค่า DB ใน .env (ใช้งานได้ แต่เปลี่ยนรหัส DB แล้ว token ทุกใบหมดอายุ)
AUTH_JWT_SECRET=

# ระดับการบังคับตัวตน/สิทธิ์:
#   off     = ไม่บังคับ
#   warn    = (ค่าเริ่มต้น) ปล่อยผ่านแต่ log เตือนเมื่อไม่มี token / ไม่มีสิทธิ์  → ใช้ช่วงแรกหลัง deploy ดูว่าทุกหน้าส่ง token ครบ
#   enforce = ไม่มี token → 401, ไม่มีสิทธิ์ → 403, แก้ค่า Setting ที่ล็อกโดยไม่ปลดล็อก → 403
AUTH_MODE=warn

# อายุ token (ชั่วโมง) ค่าเริ่มต้น 12; หน้าเว็บต่ออายุเองได้ต่อเนื่องไม่เกิน 7 วัน
# AUTH_TOKEN_HOURS=12

# อยู่หลัง nginx ให้ใส่จำนวน proxy (เช่น 1) ถ้าไม่ตั้งจะเชื่อ proxy ที่อยู่ในเครือข่ายภายในทั้งหมด
# TRUST_PROXY=1

# ขีดจำกัดต่อผู้ใช้ต่อ 15 นาที (ค่าเริ่มต้น 3000), ล็อกอินผิดต่อ (IP+รหัสพนักงาน) (20), signup/forgot-password ต่อ IP (30)
# RATE_LIMIT_MAX=3000
# LOGIN_RATE_LIMIT=20
# PUBLIC_AUTH_RATE_LIMIT=30

# ── Observability ─────────────────────────────────────────────────────────────
# /metrics (รูปแบบ Prometheus) — ถ้าตั้ง METRICS_KEY ต้องส่ง header x-metrics-key; ไม่ตั้ง = เปิดเฉพาะ IP ภายใน
# METRICS_KEY=
# เกณฑ์ log: request ช้า (ms), query ช้า (ms)
# SLOW_REQUEST_MS=3000
# SLOW_QUERY_MS=1500
# LOG_ALL_4XX=false
# แจ้ง LINE เมื่อระบบผิดปกติ (pool เต็ม, DB ล่ม, 5xx พุ่ง, event loop ค้าง) — ใช้กลุ่มแยกจากกลุ่ม delay; ไม่ตั้ง = เขียน log เท่านั้น
# OPS_ALERT_LINE_GROUP_ID=
# ALERT_POOL_PENDING=10
# ALERT_5XX_RATE=0.05
# ALERT_EVENTLOOP_LAG_MS=1500
